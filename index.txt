    <script>
        // บันทึกข้อมูล
        function submitOrder(event) {
            event.preventDefault();
            
            const name = document.getElementById('custName').value;
            const phone = document.getElementById('custPhone').value;
            const size = document.getElementById('custSize').value;
            const qty = parseInt(document.getElementById('custQty').value);
            const note = document.getElementById('custNote').value;
            const date = new Date().toLocaleString('th-TH');

            const newOrder = { date, name, phone, size, qty, note };

            // ดึงข้อมูลเดิมจาก LocalStorage
            let orders = JSON.parse(localStorage.getItem('poloOrders')) || [];
            orders.push(newOrder);

            // บันทึกกลับ
            localStorage.setItem('poloOrders', JSON.stringify(orders));

            alert('✅ บันทึกการสั่งจองสำเร็จ!\nขอบคุณที่ร่วมสั่งจองเสื้อเพชรมหาไถ่ครับ');
            document.getElementById('formOrder').reset();
        }

        // เปิดระบบแอดมิน
        function openAdmin() {
            const password = prompt('กรุณาใส่รหัสผ่านเพื่อเข้าสู่ระบบ Admin:');
            if (password === 'admin') {
                document.getElementById('main-content').classList.add('no-print', 'hidden');
                document.getElementById('admin-panel').classList.remove('hidden');
                loadAdminData();
            } else if (password !== null) {
                alert('❌ รหัสผ่านไม่ถูกต้อง');
            }
        }

        // ปิดระบบแอดมิน
        function closeAdmin() {
            document.getElementById('admin-panel').classList.add('hidden');
            document.getElementById('main-content').classList.remove('no-print', 'hidden');
        }

        // โหลดข้อมูลแสดงในตารางแอดมิน
        function loadAdminData() {
            const orders = JSON.parse(localStorage.getItem('poloOrders')) || [];
            const orderTable = document.getElementById('orderTableBody');
            const summaryTable = document.getElementById('summaryTableBody');
            
            orderTable.innerHTML = '';
            
            // สรุปยอด
            const summary = { SS: 0, S: 0, M: 0, L: 0, XL: 0, '2XL': 0, '3XL': 0, '4XL': 0, Total: 0 };

            if (orders.length === 0) {
                orderTable.innerHTML = '<tr><td colspan="8" class="text-center p-4 text-gray-500">ยังไม่มีข้อมูลการสั่งจอง</td></tr>';
            } else {
                orders.forEach((order, index) => {
                    // คำนวณสรุป
                    if(summary[order.size] !== undefined) {
                        summary[order.size] += order.qty;
                        summary.Total += order.qty;
                    }

                    // สร้างแถวข้อมูลตาราง
                    const tr = document.createElement('tr');
                    tr.innerHTML = `
                        <td class="border p-2 text-center">${index + 1}</td>
                        <td class="border p-2 text-sm">${order.date}</td>
                        <td class="border p-2">${order.name}</td>
                        <td class="border p-2">${order.phone}</td>
                        <td class="border p-2 text-center font-bold">${order.size}</td>
                        <td class="border p-2 text-center">${order.qty}</td>
                        <td class="border p-2 text-sm">${order.note}</td>
                        <td class="border p-2 text-center no-print">
                            <button onclick="deleteOrder(${index})" class="text-red-500 hover:text-red-700">ลบ</button>
                        </td>
                    `;
                    orderTable.appendChild(tr);
                });
            }

            // แสดงสรุปยอด
            summaryTable.innerHTML = `
                <tr>
                    <td class="border p-2 font-bold">${summary.SS}</td>
                    <td class="border p-2 font-bold">${summary.S}</td>
                    <td class="border p-2 font-bold">${summary.M}</td>
                    <td class="border p-2 font-bold">${summary.L}</td>
                    <td class="border p-2 font-bold">${summary.XL}</td>
                    <td class="border p-2 font-bold">${summary['2XL']}</td>
                    <td class="border p-2 font-bold">${summary['3XL']}</td>
                    <td class="border p-2 font-bold">${summary['4XL']}</td>
                    <td class="border p-2 bg-yellow-100 font-bold text-lg">${summary.Total} ตัว</td>
                </tr>
            `;
        }

        // ลบข้อมูลรายคน
        function deleteOrder(index) {
            if(confirm('คุณแน่ใจหรือไม่ว่าต้องการลบข้อมูลลำดับนี้?')) {
                let orders = JSON.parse(localStorage.getItem('poloOrders')) || [];
                orders.splice(index, 1);
                localStorage.setItem('poloOrders', JSON.stringify(orders));
                loadAdminData();
            }
        }

        // ล้างข้อมูลทั้งหมด
        function clearAllData() {
            if(confirm('⚠️ คำเตือน: คุณต้องการลบข้อมูลการสั่งจองทั้งหมดใช่หรือไม่? ข้อมูลจะไม่สามารถกู้คืนได้')) {
                localStorage.removeItem('poloOrders');
                loadAdminData();
                alert('ลบข้อมูลทั้งหมดเรียบร้อยแล้ว');
            }
        }
    </script>
